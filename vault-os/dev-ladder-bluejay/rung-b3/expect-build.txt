# b3 USB: no host feature (gs101 diff), and handheld_core_hardware does not declare it either.
absent VENDOR/etc/permissions/android.hardware.usb.host.xml
present VENDOR/etc/permissions/android.hardware.usb.accessory.xml
nogrep VENDOR/etc/permissions/handheld_core_hardware.xml android\.hardware\.usb\.host
# Gadget function instances: AOA (kernel + userspace) and, debuggable only, adb. Nothing else.
grep VENDOR/etc/init/hw/init.gs101.usb.rc ^\s+mkdir /config/usb_gadget/g1/functions/accessory\.gs2$
grep VENDOR/etc/init/hw/init.gs101.usb.rc ^\s+mkdir /config/usb_gadget/g1/functions/ffs\.aoa$
grep VENDOR/etc/init/hw/init.gs101.usb.rc ^on early-boot && property:ro\.debuggable=1$
nogrep VENDOR/etc/init/hw/init.gs101.usb.rc functions/(mtp\.gs0|ptp\.gs1|audio_source|rndis|midi|acm\.|dm\.gs7|uts\.gs8|ncm\.gs9|etr_miu|uvc\.0|ffs\.mtp|ffs\.ptp)
nogrep VENDOR/etc/init/hw/init.gs101.usb.rc ^\s+chown system system /config/usb_gadget/(g1/functions)?$
# Type-C: pinned sink at boot and boot_completed; role nodes and the OTG id node root-only.
grep VENDOR/etc/init/hw/init.gs101.usb.rc ^\s+write /sys/class/typec/port0/port_type sink$
grep VENDOR/etc/init/hw/init.gs101.usb.rc ^on property:sys\.boot_completed=1\n\s+write /sys/class/typec/port0/port_type sink
grep VENDOR/etc/ueventd.rc ^/sys/class/typec/port0 port_type\s+644 root root$
grep VENDOR/etc/ueventd.rc ^/sys/class/typec/port0 power_role\s+644 root root$
grep VENDOR/etc/ueventd.rc ^/sys/class/typec/port0 data_role\s+644 root root$
grep VENDOR/etc/ueventd.rc dwc3_exynos_otg_id\s+644 root root$
grep VENDOR/build.prop ^ro\.vendor\.phonevault\.offline=true$
